contestant.hpp

```cpp
#pragma once
#include <iomanip>
#include <iostream>
#include <string>

struct Contestant {
    long        id;      // 学号
    std::string name;    // 姓名
    std::string major;   // 专业
    int         solved;  // 解题数
    int         penalty; // 总罚时
};

/* 重载 << >>，假设 name/major 不含空格 */
inline std::ostream& operator<<(std::ostream& out, const Contestant& c) {
    out << std::left
        << std::setw(15) << c.id
        << std::setw(15) << c.name
        << std::setw(15) << c.major
        << std::setw(10) << c.solved
        << std::setw(10) << c.penalty;
    return out;
}

inline std::istream& operator>>(std::istream& in, Contestant& c) {
    in >> c.id >> c.name >> c.major >> c.solved >> c.penalty;
    return in;
}
```

utils.hpp

```cpp
#pragma once
#include <fstream>
#include <iostream>
#include <stdexcept>
#include <string>
#include <vector>
#include "contestant.hpp"

/* ACM 排序规则：解题数降序，罚时升序 */
inline bool cmp_by_solve(const Contestant& a, const Contestant& b) {
    if (a.solved != b.solved) return a.solved > b.solved;
    return a.penalty < b.penalty;
}

/* 将结果写入任意输出流 */
inline void write(std::ostream& os, const std::vector<Contestant>& v) {
    for (const auto& x : v) os << x << '\n';
}

inline void print(const std::vector<Contestant>& v) { write(std::cout, v); }

inline void save(const std::string& filename, const std::vector<Contestant>& v) {
    std::ofstream os(filename);
    if (!os) throw std::runtime_error("fail to open " + filename);
    write(os, v);
}

/* 从文件加载，跳过标题行 */
inline std::vector<Contestant> load(const std::string& filename) {
    std::ifstream is(filename);
    if (!is) throw std::runtime_error("fail to open " + filename);

    std::string line;
    std::getline(is, line); // 跳过表头

    std::vector<Contestant> v;
    Contestant t;
    int seq;
    while (is >> seq >> t) v.push_back(t);
    return v;
}
```

#### 4.1.3 问题回答

**问题 1：流操作与代码复用**
（1）`write(std::ostream&)` 能同时接收 `std::cout` 和 `std::ofstream`，因为二者都继承自同一基类 `std::ostream`，**多态引用**使得函数无需关心具体设备。
（2）若新设备也实现了 `std::ostream` 接口（如 `std::ostringstream`、自定义流缓冲区），则**无需改动** `write()`，直接传入即可。

**问题 2：异常处理与捕获**
两处 `throw`：
① `save()` 中打开文件失败时 `throw std::runtime_error(...)`；
② `load()` 中打开文件失败时同样抛出。
异常被 `app()` 的 `try/catch` 捕获，仅打印 `e.what()` 后正常结束程序。

**问题 3：lambda 替代写法**
可以替换，功能、结果、性能均一致。
lambda 等价于 `cmp_by_solve` 函数指针，符合 `std::sort` 的 `Compare` 要求。

**问题 4：数据完整性与健壮性**
（1）使用 `data_bad.txt`（含空行或字段缺失）→ `operator>>` 读取失败 → 后续所有记录无法解析 → 程序**静默生成空排行榜**。
（2）改进思路：

- 按行读取，行号计数；
- 使用 `std::istringstream` 解析，若字段不足或格式非法则 `continue` 并提示行号；
- 仅把合法行加入 `vector`。

```cpp
class Student {
```

stumgr.cpp

```cpp
#include "stumgr.hpp"
#include <fstream>
#include <iostream>
#include <algorithm>
#include <stdexcept>

void StuMgr::load(const std::string& file) {
    std::ifstream ifs(file);
    if (!ifs) throw std::runtime_error("cannot open file:" + file);

    students.clear();
    Student tmp;
    while (ifs >> tmp) students.push_back(tmp);
}

void StuMgr::sort() {
    std::stable_sort(students.begin(), students.end(),
                     [](const Student& a, const Student& b) {
                         if (a.get_major() != b.get_major())
                             return a.get_major() < b.get_major();
                         return a.get_grade() > b.get_grade();
                     });
}

void StuMgr::write(std::ostream& os) const {
    for (const auto& s : students) os << s << '\n';
}

void StuMgr::print() const { write(std::cout); }

void StuMgr::save(const std::string& file) const {
    std::ofstream ofs(file);
    if (!ofs) throw std::runtime_error("cannot open file:" + file);
    write(ofs);
}
```

#### 4.2.4 拓展（选做）：数据完整性校验

改进 `load()`，支持**报告错误行号并跳过脏数据**，效果示例：

```
Warning] line 4 format error,skipped:1003
加载成功
```

核心改动（仅展示关键差异）：

stumgr.cpp（片段）

```cpp
void StuMgr::load(const std::string& file) {
    std::ifstream ifs(file);
    if (!ifs) throw std::runtime_error("cannot open file:" + file);

    students.clear();
    std::string line;
    int lineno = 0;
    while (std::getline(ifs, line)) {
        ++lineno;
        if (line.empty()) continue;
        std::istringstream iss(line);
        Student tmp;
        if (!(iss >> tmp)) {
            std::cerr << "[Warning] line " << lineno
                      << " format error,skipped:" << line << '\n';
            continue;
        }
        if (tmp.get_grade() < 0 || tmp.get_grade() > 100) {
            std::cerr << "[Warning] line " << lineno
                      << " invalid grade,skipped:" << line << '\n';
            continue;
        }
        students.push_back(tmp);
    }
}
```

## 五、实验总结（选写）

1. 通过任务 1 体会到“**面向接口编程**”的优雅：`write(std::ostream&)` 一处代码覆盖屏幕、文件、内存字符串等多种设备。
